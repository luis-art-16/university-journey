SOURCE SPORTING.txt;				/* Executar o source SPORTING.txt */

USE SPORTING;					/* Utilizar a base de dados SPORTING */

n_jogador INT,
nome CHAR(25),
);

n_jogador INT NOT NULL,				/* Nao aceita que o campo n_jogador seja nulo na tabela */

N_MODALIDADE int not null auto_increment,	/* Incrementa automáticamente o campo N_MODALIDADE com inteiros */

primary key (n_jogador)				/* Define que o campo n_jogador é a chave primária nesta tabela */

unique (nome)					/* Define que o campo nome vai ser chave secundária nesta tabela e não se vai repetir */
