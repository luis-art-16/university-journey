"""Obstacle avoiding controller."""

# Importa a classe Robot do módulo controller
from controller import Robot

# Constantes globais
TIMESTEP = 32
MAX_SPEED = 6.28

def run_robot(robot):
    # Obtém os dispositivos do robô
    try:
        left_motor = robot.getDevice("left wheel motor")
        right_motor = robot.getDevice("right wheel motor")
    except Exception as e:
        print(f"Erro ao obter os dispositivos do robô: {e}")
        return

    # Configura os motores para modo de velocidade infinita
    left_motor.setPosition(float('inf'))
    right_motor.setPosition(float('inf'))

    # Inicializa as velocidades dos motores
    left_motor.setVelocity(0.0)
    right_motor.setVelocity(0.0)
    
    #ir0 = robot.getDevice('ps0')
    #ir0.enable(TIMESTEP)
    
    list_ps = []
    for ind in [0,1,2,5,6,7]:
        sensor_name = 'ps' + str(ind)
        list_ps.append(robot.getDevice(sensor_name))
        list_ps[-1].enable(TIMESTEP)
        

    # Loop principal
    while robot.step(TIMESTEP) != -1:
        # Comandos de controle dos motores
        left_speed = MAX_SPEED
        right_speed = MAX_SPEED
        
        
        for ps in list_ps:
            ps_val = ps.getValue()
            print(ps_val)
            if ps_val > 100:
               #true 
               left_speed = -MAX_SPEED
        
        
        
        left_motor.setVelocity(left_speed)
        right_motor.setVelocity(right_speed)

# Bloco principal
if __name__ == "__main__":
    # Cria a instância do robô
    robot = Robot()
    run_robot(robot)
